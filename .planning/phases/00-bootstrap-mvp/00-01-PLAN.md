    ```typescript
    import { execFileSync } from 'child_process';
    import { mkdtempSync, rmSync, cpSync } from 'fs';
    import { join } from 'path';
    import os from 'os';
    import path from 'path';

    export const TOOLS_PATH = path.resolve(__dirname, '../terrace-tools.cjs');
    export const FIXTURES_DIR = path.resolve(__dirname, '../fixtures');

    export interface TerraceResult {
      stdout: string;
      stderr: string;
      code: number;
    }

    export function runTerrace(args: string[], cwd: string): TerraceResult {
      try {
        const stdout = execFileSync(process.execPath, [TOOLS_PATH, ...args], {
          cwd,
          encoding: 'utf8',
        });
        return { stdout, stderr: '', code: 0 };
      } catch (err: any) {
        return { stdout: err.stdout ?? '', stderr: err.stderr ?? '', code: err.status ?? 1 };
      }
    }

    export function copyFixture(fixtureName: string): string {
      const tmpDir = mkdtempSync(join(os.tmpdir(), 'terrace-test-'));
      cpSync(join(FIXTURES_DIR, fixtureName), tmpDir, { recursive: true });
      return tmpDir;
    }

    export function makeTmpDir(): string {
      return mkdtempSync(join(os.tmpdir(), 'terrace-test-'));
    }

    export function cleanupDir(dir: string): void {
      rmSync(dir, { recursive: true, force: true });
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| test → CLI subprocess | Tests spawn terrace-tools.cjs via execFileSync; args array passed directly |
| test → tmpDir filesystem | Tests write to OS tmpdir, never to git-tracked fixtures/ |
| --spec-ref arg → CLI | User-supplied spec_ref value; must be validated for path traversal in later plans |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-00-01 | Tampering | tests/fixtures/ directory | mitigate | Tests MUST copy fixtures to tmpDir (mkdtempSync) before CLI operations — never write to git-tracked fixtures/ directory; documented in Pitfall 4 of RESEARCH.md |
| T-00-02 | Information Disclosure | test tmpDir | mitigate | afterEach calls `cleanupDir(tmpDir)` via `fs.rmSync(dir, { recursive: true, force: true })` to remove all test artifacts |
| T-00-03 | Elevation of Privilege | package.json scripts | accept | Dev-only toolchain; no production attack surface; scripts run only on developer machines |
| T-00-04 | Spoofing | RED test assertions | mitigate | Tests must assert on specific exit code and stdout content — not just `result.code !== undefined`; prevents tests that pass vacuously |
</threat_model>
