```javascript
'use strict';

const fs = require('fs');
const path = require('path');
const { parseFrontmatter, loadConfig } = require('./core.cjs');

/**
 * @typedef {{ code: string; message: string; file?: string }} ValidationIssue
 * @typedef {{ blocking: ValidationIssue[]; warnings: ValidationIssue[] }} ValidationResult
 * @typedef {{ file_mapping?: Record<string, string> }} ValidationConfig
 */

/**
 * REQUIRED_SECTIONS — sections/fields that must appear in each artifact (as plain-text substrings).
 * Blocking errors are raised if any listed string is absent.
 */
const REQUIRED_SECTIONS = {
  'PRD.md': ['problem', 'actors', 'desired_outcomes', 'non_goals', 'constraints', 'success_criteria', 'open_questions'],
  'COMPILED-SPEC.md': ['spec_version', 'project', 'phase', 'requirements', 'protected', 'last_updated', 'source_refs'],
};

/** Default file locations relative to repo root */
const DEFAULT_FILE_LOCATIONS = {
  'PRD.md': 'docs/prd/PRD.md',
  'COMPILED-SPEC.md': 'docs/spec/COMPILED-SPEC.md',
  'DECISION-LOG.md': 'docs/DECISION-LOG.md',
};

/** STALE_SESSION threshold: 30 days in milliseconds */
const STALE_SESSION_THRESHOLD_MS = 30 * 24 * 60 * 60 * 1000;

/**
 * validateArtifacts(cwd, config) — validates governance artifacts in the repo.
 * Returns blocking errors and warnings. Never throws.
 * @param {string} cwd
 * @param {ValidationConfig} config
 * @returns {ValidationResult}
 */
function validateArtifacts(cwd, config) {
  const resolvedCwd = path.resolve(cwd);
  const fileMapping = (config && config.file_mapping) ? config.file_mapping : {};

  /** @type {ValidationIssue[]} */
  const blocking = [];
  /** @type {ValidationIssue[]} */
  const warnings = [];

  /**
   * resolveArtifactPath — uses custom mapping if provided, else falls back to default location.
   */
  function resolveArtifactPath(name) {
    const mapped = fileMapping[name] || DEFAULT_FILE_LOCATIONS[name];
    if (!mapped) return null;
    return path.resolve(resolvedCwd, mapped);
  }

  /**
   * checkRequiredSections — reads a file and checks each required string is present.
   * Produces a blocking error for each missing string.
   */
  function checkRequiredSections(filePath, required) {
    if (!fs.existsSync(filePath)) return; // Missing file is not this validator's concern
    let content;
    try {
      content = fs.readFileSync(filePath, 'utf-8');
    } catch (err) {
      blocking.push({ code: 'FILE_READ_ERROR', message: `Cannot read ${filePath}: ${err.message}`, file: filePath });
      return;
    }
    for (const section of required) {
      if (!content.includes(section)) {
        blocking.push({
          code: 'MISSING_REQUIRED_SECTION',
          message: `${path.basename(filePath)} is missing required section or field: '${section}'`,
          file: filePath
        });
      }
    }
  }

  // Check PRD.md sections
  const prdPath = resolveArtifactPath('PRD.md');
  if (prdPath) checkRequiredSections(prdPath, REQUIRED_SECTIONS['PRD.md']);

  // Check COMPILED-SPEC.md frontmatter fields
  const specPath = resolveArtifactPath('COMPILED-SPEC.md');
  if (specPath) checkRequiredSections(specPath, REQUIRED_SECTIONS['COMPILED-SPEC.md']);

  // Check DECISION-LOG.md for missing spec_ref fields
  const decisionLogPath = resolveArtifactPath('DECISION-LOG.md');
  if (decisionLogPath && fs.existsSync(decisionLogPath)) {
    try {
      const content = fs.readFileSync(decisionLogPath, 'utf-8');
      // Check each YAML frontmatter block for spec_ref
      // A DECISION-LOG.md may have multiple entry blocks separated by ---
      const blocks = content.split(/^---$/m);
      let foundMissingSpecRef = false;
      for (const block of blocks) {
        if (block.trim().length === 0) continue;
        // If block contains decision_id but not spec_ref, it's a missing reference
        if (block.includes('decision_id:') && !block.includes('spec_ref:')) {
          foundMissingSpecRef = true;
          break;
        }
      }
      if (foundMissingSpecRef) {
        blocking.push({
          code: 'MISSING_SPEC_REF',
          message: `DECISION-LOG.md contains an entry missing 'spec_ref' field`,
          file: decisionLogPath
        });
      }
    } catch (err) {
      blocking.push({ code: 'FILE_READ_ERROR', message: `Cannot read DECISION-LOG.md: ${err.message}`, file: decisionLogPath });
    }
  }

  // Check project-state.json for stale session (VAL-04)
  const stateFile = path.join(resolvedCwd, '.terrace', 'project-state.json');
  if (fs.existsSync(stateFile)) {
    try {
      const state = JSON.parse(fs.readFileSync(stateFile, 'utf-8'));
      if (state.last_session) {
        const lastSessionDate = new Date(state.last_session);
        const age = Date.now() - lastSessionDate.getTime();
        if (!isNaN(lastSessionDate.getTime()) && age > STALE_SESSION_THRESHOLD_MS) {
          warnings.push({
            code: 'STALE_SESSION',
            message: `project-state.json.last_session is older than 30 days (last: ${state.last_session}). Run: terrace session start`,
          });
        }
      }
    } catch (_) {
      // Invalid JSON — already caught by doctor; skip here
    }
  }

  return { blocking, warnings };
}

module.exports = { validateArtifacts };
```
