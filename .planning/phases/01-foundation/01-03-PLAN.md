```javascript
'use strict';

const fs = require('fs');
const path = require('path');

/**
 * output(data, jsonMode) — all stdout goes through this function.
 * In jsonMode: JSON.stringify(data, null, 2) + newline.
 * In human mode: if data is a string, write as-is; if array/object, write JSON (same content, same function).
 * Per CLI-12: --json flag controls jsonMode; all commands use the same output function.
 * @param {unknown} data
 * @param {boolean} [jsonMode=false]
 */
function output(data, jsonMode = false) {
  if (jsonMode || typeof data !== 'string') {
    process.stdout.write(JSON.stringify(data, null, 2) + '\n');
  } else {
    process.stdout.write(String(data) + '\n');
  }
}

/**
 * error(msg) — writes to stderr and exits 1.
 * @param {string} msg
 * @returns {never}
 */
function error(msg) {
  process.stderr.write(msg + '\n');
  process.exit(1);
}

/**
 * resolveTerraceDir(cwd) — returns the absolute path of .terrace/ for the given working directory.
 * @param {string} cwd
 * @returns {string}
 */
function resolveTerraceDir(cwd) {
  return path.resolve(cwd, '.terrace');
}

/**
 * isAllowedWritePath(cwd, targetPath) — returns true if targetPath is under {cwd}/.terrace or {cwd}/docs.
 * Used to enforce GSD isolation (D-06, GSD-04).
 * @param {string} cwd
 * @param {string} targetPath
 * @returns {boolean}
 */
function isAllowedWritePath(cwd, targetPath) {
  const resolved = path.resolve(targetPath);
  const terraceRoot = path.resolve(cwd, '.terrace');
  const docsRoot = path.resolve(cwd, 'docs');
  return resolved.startsWith(terraceRoot) || resolved.startsWith(docsRoot);
}

/**
 * loadConfig(cwd) — reads .terrace/policy.json if it exists; returns {} if not.
 * Never throws — returns empty config on any read/parse error.
 * @param {string} cwd
 * @returns {Record<string, unknown>}
 */
function loadConfig(cwd) {
  const policyPath = path.join(path.resolve(cwd, '.terrace'), 'policy.json');
  try {
    return JSON.parse(fs.readFileSync(policyPath, 'utf-8'));
  } catch (_) {
    return {};
  }
}

/**
 * parseFrontmatter(content) — extracts YAML frontmatter from Markdown content.
 * Returns the frontmatter string (between --- delimiters) and the body.
 * Follows the gsd-tools.cjs frontmatter.cjs pattern: stdlib regex, no external YAML parser.
 * @param {string} content
 * @returns {{ frontmatter: string; body: string }}
 */
function parseFrontmatter(content) {
  const match = content.match(/^---\n([\s\S]*?)\n---\n?([\s\S]*)$/);
  if (!match) return { frontmatter: '', body: content };
  return { frontmatter: match[1], body: match[2] };
}

module.exports = { output, error, resolveTerraceDir, isAllowedWritePath, loadConfig, parseFrontmatter };
```

--- src/lib/init.cjs ---

Key behaviors (derived from tests/init.test.ts):
1. cmdInit(cwd, options) returns a ManifestEntry array: `{ file: string, action: 'created'|'skipped'|'overwritten' }[]`
2. Every write is checked against isAllowedWritePath before execution
3. If options.force is false and a file exists: push { file, action: 'skipped' } and continue
4. If options.force is true and a file exists: overwrite and push { file, action: 'overwritten' }
5. If file does not exist: create and push { file, action: 'created' }
6. Directories are created with `fs.mkdirSync(dir, { recursive: true })` before file writes
7. No file is written to .planning/ or any path outside .terrace/ or docs/

Files to install in a clean repo:
- `{cwd}/.terrace/project-state.json` — initial content: `{ "phase": "intake", "spec_hash": null, "active_slice": null, "last_session": null, "policy_mode": "standard" }`
- `{cwd}/.terrace/presets/registry.json` — initial content: `{ "version": "1.0", "presets": [] }`
- `{cwd}/.terrace/policy.json` — initial content: `{}`
- `{cwd}/docs/prd/.gitkeep` — empty file (creates directory)
- `{cwd}/docs/spec/.gitkeep` — empty file (creates directory)
- `{cwd}/docs/testing/.gitkeep` — empty file (creates directory)

```javascript
'use strict';

const fs = require('fs');
const path = require('path');
const { isAllowedWritePath } = require('./core.cjs');

/** @typedef {{ file: string; action: 'created' | 'skipped' | 'overwritten' }} ManifestEntry */

/**
 * writeWithManifest — writes a file if allowed; records action in results.
 * @param {string} filePath - Absolute path
 * @param {string} content - File content
 * @param {boolean} force
 * @param {string} cwd - Repo root for path allowlist check
 * @param {ManifestEntry[]} results
 */
function writeWithManifest(filePath, content, force, cwd, results) {
  if (!isAllowedWritePath(cwd, filePath)) {
    throw new Error(`Security violation: attempted write outside allowed paths: ${filePath}`);
  }
  fs.mkdirSync(path.dirname(filePath), { recursive: true });
  const exists = fs.existsSync(filePath);
  if (exists && !force) {
    results.push({ file: filePath, action: 'skipped' });
    return;
  }
  fs.writeFileSync(filePath, content, 'utf-8');
  results.push({ file: filePath, action: exists ? 'overwritten' : 'created' });
}

/**
 * cmdInit — installs Terrace artifacts into target repo.
 * @param {string} cwd - Target repo root
 * @param {{ force: boolean; yes: boolean }} options
 * @returns {ManifestEntry[]}
 */
function cmdInit(cwd, options) {
  const resolvedCwd = path.resolve(cwd);
  const { force } = options;
  /** @type {ManifestEntry[]} */
  const results = [];

  const initialState = JSON.stringify({
    phase: 'intake',
    spec_hash: null,
    active_slice: null,
    last_session: null,
    policy_mode: 'standard'
  }, null, 2);

  const initialRegistry = JSON.stringify({ version: '1.0', presets: [] }, null, 2);
  const initialPolicy = JSON.stringify({}, null, 2);

  const writes = [
    { file: path.join(resolvedCwd, '.terrace', 'project-state.json'), content: initialState },
    { file: path.join(resolvedCwd, '.terrace', 'presets', 'registry.json'), content: initialRegistry },
    { file: path.join(resolvedCwd, '.terrace', 'policy.json'), content: initialPolicy },
    { file: path.join(resolvedCwd, 'docs', 'prd', '.gitkeep'), content: '' },
    { file: path.join(resolvedCwd, 'docs', 'spec', '.gitkeep'), content: '' },
    { file: path.join(resolvedCwd, 'docs', 'testing', '.gitkeep'), content: '' },
  ];

  for (const { file, content } of writes) {
    writeWithManifest(file, content, force, resolvedCwd, results);
  }

  return results;
}

module.exports = { cmdInit };
```
