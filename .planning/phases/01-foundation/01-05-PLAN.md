```javascript
'use strict';

/**
 * LEGAL_TRANSITIONS — hardcoded phase transition table (D-12).
 * Each key is a valid from-phase; value is the array of valid to-phases.
 * Externalization to JSON config is deferred to a later phase.
 * @type {Record<string, string[]>}
 */
const LEGAL_TRANSITIONS = {
  'intake':               ['interrogation'],
  'interrogation':        ['spec-compilation', 'intake'],
  'spec-compilation':     ['test-architecture', 'interrogation'],
  'test-architecture':    ['baseline-protection', 'spec-compilation'],
  'baseline-protection':  ['implementation', 'test-architecture'],
  'implementation':       ['adversarial-review', 'baseline-protection'],
  'adversarial-review':   ['regression-capture', 'implementation'],
  'regression-capture':   ['handoff', 'adversarial-review'],
  'handoff':              ['intake'],
};

/**
 * isLegalTransition — returns true if transitioning from -> to is in the legal table.
 * @param {string} from
 * @param {string} to
 * @returns {boolean}
 */
function isLegalTransition(from, to) {
  const allowed = LEGAL_TRANSITIONS[from];
  if (!allowed) return false;
  return allowed.includes(to);
}

/**
 * setPhase — returns a new state object with phase updated to newPhase.
 * Throws if the transition is illegal.
 * @param {Record<string, unknown>} state - Current project-state object
 * @param {string} newPhase
 * @returns {Record<string, unknown>}
 * @throws {Error} if transition is illegal
 */
function setPhase(state, newPhase) {
  const currentPhase = String(state.phase);
  if (!isLegalTransition(currentPhase, newPhase)) {
    throw new Error(
      `illegal transition: cannot move from '${currentPhase}' to '${newPhase}'. ` +
      `Legal transitions from '${currentPhase}': ${(LEGAL_TRANSITIONS[currentPhase] || []).join(', ') || 'none'}`
    );
  }
  return { ...state, phase: newPhase };
}

module.exports = { isLegalTransition, setPhase, LEGAL_TRANSITIONS };
```

Key behaviors (from tests/preset.test.ts):
1. installPreset adds entry with id, name, version, installed_at (ISO timestamp), conflicts (empty array)
2. installPreset is idempotent: same id + same version = skips and returns { conflict: false, message }
3. installPreset with same id + different version = returns { conflict: true, message containing id }; does NOT overwrite
4. listPresets returns the presets array from registry.json
5. Flags written to policy.json under key equal to preset id (namespaced, PRST-13): existing keys preserved

```javascript
'use strict';

const fs = require('fs');
const path = require('path');
const { loadConfig } = require('./core.cjs');

/**
 * @typedef {{ id: string; name: string; version: string; category: string; effect: string;
 *             agents: string[]; workflows: string[]; fragments: string[];
 *             flags: Array<{key: string; type: string; default: unknown}> }} PresetManifest
 * @typedef {{ conflict: boolean; message: string }} InstallResult
 */

/**
 * readRegistry(terraceDir) — reads .terrace/presets/registry.json.
 * Returns { version: '1.0', presets: [] } if not found.
 * @param {string} terraceDir
 * @returns {{ version: string; presets: Array<Record<string, unknown>> }}
 */
function readRegistry(terraceDir) {
  const reg = path.join(terraceDir, 'presets', 'registry.json');
  if (!fs.existsSync(reg)) return { version: '1.0', presets: [] };
  try {
    return JSON.parse(fs.readFileSync(reg, 'utf-8'));
  } catch (_) {
    return { version: '1.0', presets: [] };
  }
}

/**
 * writeRegistry(terraceDir, data) — writes registry atomically.
 * @param {string} terraceDir
 * @param {{ version: string; presets: Array<Record<string, unknown>> }} data
 */
function writeRegistry(terraceDir, data) {
  const dir = path.join(terraceDir, 'presets');
  fs.mkdirSync(dir, { recursive: true });
  fs.writeFileSync(path.join(dir, 'registry.json'), JSON.stringify(data, null, 2), 'utf-8');
}

/**
 * installPreset(cwd, manifest, options) — registers a preset in registry.json.
 * Per D-14: scaffold-only in Phase 1 (registers entry + conflict check; no capability loading).
 * @param {string} cwd
 * @param {PresetManifest} manifest
 * @param {{ force: boolean }} options
 * @returns {InstallResult}
 */
function installPreset(cwd, manifest, options) {
  const terraceDir = path.resolve(cwd, '.terrace');
  const registry = readRegistry(terraceDir);

  const existing = registry.presets.find((p) => p.id === manifest.id);

  if (existing) {
    if (existing.version === manifest.version) {
      // Idempotent: same version already installed
      return { conflict: false, message: `${manifest.id}@${manifest.version} already installed` };
    } else {
      // Conflict: different version
      return {
        conflict: true,
        message: `Conflict: ${manifest.id} is already installed at version ${existing.version}. ` +
                 `Requested version: ${manifest.version}. Use --force to overwrite.`
      };
    }
  }

  // New install: add entry
  const entry = {
    id: manifest.id,
    name: manifest.name,
    version: manifest.version,
    category: manifest.category,
    effect: manifest.effect,
    installed_at: new Date().toISOString(),
    conflicts: [],
    agents: manifest.agents,
    workflows: manifest.workflows,
    fragments: manifest.fragments,
  };

  registry.presets.push(entry);
  writeRegistry(terraceDir, registry);

  // Write flags to policy.json under namespaced key (PRST-13)
  if (manifest.flags && manifest.flags.length > 0) {
    const policyPath = path.join(terraceDir, 'policy.json');
    let policy = {};
    if (fs.existsSync(policyPath)) {
      try { policy = JSON.parse(fs.readFileSync(policyPath, 'utf-8')); } catch (_) { policy = {}; }
    }
    const flagDefaults = {};
    for (const flag of manifest.flags) {
      flagDefaults[flag.key] = flag.default;
    }
    policy[manifest.id] = flagDefaults;
    fs.writeFileSync(policyPath, JSON.stringify(policy, null, 2), 'utf-8');
  }

  return { conflict: false, message: `${manifest.id}@${manifest.version} installed` };
}

/**
 * listPresets(cwd) — returns all installed presets.
 * @param {string} cwd
 * @returns {Array<Record<string, unknown>>}
 */
function listPresets(cwd) {
  const terraceDir = path.resolve(cwd, '.terrace');
  return readRegistry(terraceDir).presets;
}

module.exports = { installPreset, listPresets };
```
