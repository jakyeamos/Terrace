Write src/lib/spec-hash.cjs following the exact CJS pattern from src/lib/core.cjs:

```javascript
'use strict';
const fs = require('fs');
const path = require('path');
const crypto = require('crypto');

/**
 * Compute a semantic hash of a spec document, excluding non-semantic changes.
 * Strips: last_updated: lines, blank lines, HTML comment-only lines, trailing whitespace.
 * @param {string} filePath - Path to the spec file (must be within process.cwd())
 * @returns {string} SHA-256 hex digest (64 lowercase chars)
 */
function computeSpecHash(filePath) {
  const resolved = fs.realpathSync(filePath);
  const cwd = fs.realpathSync(process.cwd());
  if (!resolved.startsWith(cwd + path.sep) && resolved !== cwd) {
    throw new Error('UNSAFE_PATH: file is outside the project root');
  }
  const raw = fs.readFileSync(resolved, 'utf-8');
  const lines = raw.split('\n');
  const normalized = lines
    .filter(line => !/^last_updated:/.test(line))
    .filter(line => !/^\s*$/.test(line))
    .filter(line => !/^\s*<!--.*-->\s*$/.test(line))
    .map(line => line.trimEnd())
    .join('\n');
  return crypto.createHash('sha256').update(normalized, 'utf-8').digest('hex');
}

module.exports = { computeSpecHash };
```

});
